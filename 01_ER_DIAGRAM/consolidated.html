ER Diagram ฉบับสมบูรณ์ (Enterprise Hospital HIS v2 ER Diagram - 3NF)

province_name

[PK] branch_id

branch_name

[FK] province_id, registered_branch_id
title_prefix, first_name, last_name
gender, birth_date, blood_group
weight_kg, height_cm, patient_status
H1: patient_contacts (3NF)
[PK] contact_id
[FK] patient_id
contact_type ('Phone', 'Email')
contact_value, is_primary
H1: patient_insurance_policies
[PK] policy_id
[FK] patient_id, provider_id
policy_number, coverage_limit
expiry_date
H1: providers
[PK] provider_id
provider_name
H7: departments
[PK] department_id
[FK] branch_id

[PK] doctor_id
[FK] department_id

salary_thb

[PK] appointment_id
[FK] patient_id, doctor_id, branch_id
appointment_date, appointment_time
appointment_status, appointment_source

[PK] record_id
[FK] patient_id, doctor_id, diagnosis_id

[PK] user_id
