-- =====================================================================
-- HOSPITAL HR SYSTEM - SCHEMA (SAFE / IDEMPOTENT VERSION)
-- รันซ้ำได้กี่ครั้งก็ได้โดยไม่ error ไม่ว่าตอนนี้ schema จะมีตารางอยู่แล้ว
-- บางส่วนหรือไม่มีเลยก็ตาม -- ตารางไหนมีอยู่แล้วจะถูกข้าม ตารางไหนยังไม่มี
-- จะถูกสร้างเพิ่มให้ครบ ผลลัพธ์สุดท้ายจะได้ทั้งหมด 22 ตารางเสมอ
-- Target: PostgreSQL 13+
-- =====================================================================

-- ไม่ต้องใช้ CREATE EXTENSION pgcrypto เพราะ PostgreSQL 13+ มี
-- gen_random_uuid() ในตัวอยู่แล้ว


-- =====================================================================
-- 1) ตาราง lookup พื้นฐาน
-- =====================================================================

CREATE TABLE IF NOT EXISTS titles (
    title_id     serial PRIMARY KEY,
    title_name   character varying(30) NOT NULL UNIQUE
);

INSERT INTO titles (title_name) VALUES
    ('นาย'), ('นาง'), ('นางสาว'), ('เด็กชาย'), ('เด็กหญิง')
ON CONFLICT (title_name) DO NOTHING;

CREATE TABLE IF NOT EXISTS genders (
    gender_id     serial PRIMARY KEY,
    gender_name   character varying(20) NOT NULL UNIQUE
);

INSERT INTO genders (gender_name) VALUES
    ('ชาย'), ('หญิง'), ('ไม่ระบุ')
ON CONFLICT (gender_name) DO NOTHING;

CREATE TABLE IF NOT EXISTS shift_types (
    shift_type_id   serial PRIMARY KEY,
    shift_name      character varying(50) NOT NULL,
    start_time      time NOT NULL,
    end_time        time NOT NULL,
    description     text
);

CREATE TABLE IF NOT EXISTS specialties (
    specialty_id    serial PRIMARY KEY,
    specialty_name  character varying(150) NOT NULL,
    description     text
);

CREATE TABLE IF NOT EXISTS leave_types (
    leave_type_id     serial PRIMARY KEY,
    leave_name        character varying(50) NOT NULL,
    max_days_per_year integer,
    description       text
);

CREATE TABLE IF NOT EXISTS roles (
    role_id       serial PRIMARY KEY,
    role_name     character varying(50) NOT NULL UNIQUE,
    description   text
);


-- =====================================================================
-- 2) ตารางหลัก
-- =====================================================================

CREATE TABLE IF NOT EXISTS departments (
    department_id      serial PRIMARY KEY,
    department_name    character varying(100) NOT NULL,
    contact_number      character varying(20),
    department_head_id  uuid
);

CREATE TABLE IF NOT EXISTS positions (
    position_id     serial PRIMARY KEY,
    position_name   character varying(100) NOT NULL,
    base_salary      numeric(10,2)
);

CREATE TABLE IF NOT EXISTS employees (
    employee_id       uuid PRIMARY KEY DEFAULT gen_random_uuid(),
    title_id            integer REFERENCES titles(title_id),
    gender_id             integer REFERENCES genders(gender_id),
    first_name               character varying(100) NOT NULL,
    last_name                   character varying(100) NOT NULL,
    national_id                    character varying(13),
    date_of_birth                     date,
    hire_date                            date,
    department_id                           integer NOT NULL REFERENCES departments(department_id),
    position_id                                integer NOT NULL REFERENCES positions(position_id),
    status                                        character varying(20),
    phone                                            character varying(20),
    email                                               character varying(150) UNIQUE,
    address                                                text,
    marital_status                                            character varying(20),
    photo_url                                                    text
);

ALTER TABLE employees
    ADD COLUMN IF NOT EXISTS title_id integer REFERENCES titles(title_id),
    ADD COLUMN IF NOT EXISTS gender_id integer REFERENCES genders(gender_id),
    ADD COLUMN IF NOT EXISTS phone character varying(20),
    ADD COLUMN IF NOT EXISTS email character varying(150),
    ADD COLUMN IF NOT EXISTS address text,
    ADD COLUMN IF NOT EXISTS marital_status character varying(20),
    ADD COLUMN IF NOT EXISTS photo_url text;

DO $$
BEGIN
    IF NOT EXISTS (
        SELECT 1 FROM pg_constraint WHERE conname = 'fk_departments_head'
    ) THEN
        ALTER TABLE departments
            ADD CONSTRAINT fk_departments_head
            FOREIGN KEY (department_head_id) REFERENCES employees(employee_id);
    END IF;
END $$;

CREATE TABLE IF NOT EXISTS medical_licenses (
    license_id           serial PRIMARY KEY,
    employee_id            uuid NOT NULL REFERENCES employees(employee_id) ON DELETE CASCADE,
    license_type              character varying(100),
    license_number               character varying(50),
    issued_date                     date,
    expiry_date                        date,
    issuing_authority                     character varying(150),
    status                                    character varying(20) NOT NULL DEFAULT 'active'
);

ALTER TABLE medical_licenses
    ADD COLUMN IF NOT EXISTS issuing_authority character varying(150),
    ADD COLUMN IF NOT EXISTS status character varying(20) NOT NULL DEFAULT 'active';

CREATE TABLE IF NOT EXISTS work_shifts (
    shift_id         serial PRIMARY KEY,
    employee_id         uuid NOT NULL REFERENCES employees(employee_id) ON DELETE CASCADE,
    shift_date              date NOT NULL,
    shift_type                  character varying(20),
    shift_type_id                   integer REFERENCES shift_types(shift_type_id)
);

ALTER TABLE work_shifts
    ADD COLUMN IF NOT EXISTS shift_type_id integer REFERENCES shift_types(shift_type_id);


-- =====================================================================
-- 3) ตารางเชื่อมความสัมพันธ์และ lookup เสริม
-- =====================================================================

CREATE TABLE IF NOT EXISTS employee_specialties (
    employee_id     uuid NOT NULL REFERENCES employees(employee_id) ON DELETE CASCADE,
    specialty_id    integer NOT NULL REFERENCES specialties(specialty_id) ON DELETE CASCADE,
    certified_date  date,
    PRIMARY KEY (employee_id, specialty_id)
);


-- =====================================================================
-- 4) ประวัติการโยกย้ายแผนก/ตำแหน่ง
-- =====================================================================

CREATE TABLE IF NOT EXISTS employee_department_history (
    history_id    serial PRIMARY KEY,
    employee_id     uuid NOT NULL REFERENCES employees(employee_id) ON DELETE CASCADE,
    department_id      integer NOT NULL REFERENCES departments(department_id),
    start_date            date NOT NULL,
    end_date                 date
);

CREATE TABLE IF NOT EXISTS employee_position_history (
    history_id    serial PRIMARY KEY,
    employee_id     uuid NOT NULL REFERENCES employees(employee_id) ON DELETE CASCADE,
    position_id       integer NOT NULL REFERENCES positions(position_id),
    start_date            date NOT NULL,
    end_date                 date
);


-- =====================================================================
-- 5) การลา / การเข้างาน
-- =====================================================================

CREATE TABLE IF NOT EXISTS leave_requests (
    leave_request_id   serial PRIMARY KEY,
    employee_id           uuid NOT NULL REFERENCES employees(employee_id) ON DELETE CASCADE,
    leave_type_id            integer NOT NULL REFERENCES leave_types(leave_type_id),
    start_date                  date NOT NULL,
    end_date                       date NOT NULL,
    reason                            text,
    status                               character varying(20) NOT NULL DEFAULT 'pending',
    approved_by                            uuid REFERENCES employees(employee_id),
    requested_at                              timestamp NOT NULL DEFAULT now(),
    approved_at                                  timestamp,
    CHECK (end_date >= start_date)
);

CREATE TABLE IF NOT EXISTS attendance (
    attendance_id    serial PRIMARY KEY,
    employee_id         uuid NOT NULL REFERENCES employees(employee_id) ON DELETE CASCADE,
    shift_id               integer REFERENCES work_shifts(shift_id),
    work_date                 date NOT NULL,
    check_in_time                timestamp,
    check_out_time                  timestamp,
    status                             character varying(20) DEFAULT 'present'
);


-- =====================================================================
-- 6) สัญญาจ้าง / เงินเดือน
-- =====================================================================

CREATE TABLE IF NOT EXISTS employee_contracts (
    contract_id        serial PRIMARY KEY,
    employee_id           uuid NOT NULL REFERENCES employees(employee_id) ON DELETE CASCADE,
    contract_type            character varying(50) NOT NULL,
    start_date                  date NOT NULL,
    end_date                       date,
    probation_end_date                date,
    contract_file_url                    text,
    is_active                                boolean NOT NULL DEFAULT true
);

CREATE TABLE IF NOT EXISTS payroll (
    payroll_id         serial PRIMARY KEY,
    employee_id           uuid NOT NULL REFERENCES employees(employee_id) ON DELETE CASCADE,
    pay_period_start         date NOT NULL,
    pay_period_end              date NOT NULL,
    net_amount                     numeric(10,2) NOT NULL,
    paid_at                           timestamp,
    status                               character varying(20) NOT NULL DEFAULT 'pending'
);

CREATE TABLE IF NOT EXISTS payroll_items (
    payroll_item_id     serial PRIMARY KEY,
    payroll_id             integer NOT NULL REFERENCES payroll(payroll_id) ON DELETE CASCADE,
    item_type                 character varying(20) NOT NULL,
    item_name                    character varying(100) NOT NULL,
    amount                          numeric(10,2) NOT NULL
);


-- =====================================================================
-- 7) ข้อมูลเสริม
-- =====================================================================

CREATE TABLE IF NOT EXISTS emergency_contacts (
    contact_id        serial PRIMARY KEY,
    employee_id          uuid NOT NULL REFERENCES employees(employee_id) ON DELETE CASCADE,
    contact_name            character varying(150) NOT NULL,
    relationship               character varying(50),
    phone_number                  character varying(20) NOT NULL
);


-- =====================================================================
-- 8) ระบบผู้ใช้ / สิทธิ์การเข้าถึง / audit log
-- =====================================================================

CREATE TABLE IF NOT EXISTS users (
    user_id           uuid PRIMARY KEY DEFAULT gen_random_uuid(),
    employee_id          uuid UNIQUE REFERENCES employees(employee_id) ON DELETE CASCADE,
    username                character varying(100) NOT NULL UNIQUE,
    password_hash              text NOT NULL,
    role_id                       integer NOT NULL REFERENCES roles(role_id),
    is_active                        boolean NOT NULL DEFAULT true,
    last_login_at                       timestamp,
    created_at                             timestamp NOT NULL DEFAULT now()
);

CREATE TABLE IF NOT EXISTS audit_logs (
    audit_id          serial PRIMARY KEY,
    user_id               uuid REFERENCES users(user_id),
    action                   character varying(50) NOT NULL,
    table_name                  character varying(100) NOT NULL,
    record_id                      text,
    old_values                        jsonb,
    new_values                           jsonb,
    created_at                              timestamp NOT NULL DEFAULT now()
);


-- =====================================================================
-- 9) Indexes
-- =====================================================================

CREATE INDEX IF NOT EXISTS idx_employees_title ON employees(title_id);
CREATE INDEX IF NOT EXISTS idx_employees_gender ON employees(gender_id);
CREATE INDEX IF NOT EXISTS idx_employees_department ON employees(department_id);
CREATE INDEX IF NOT EXISTS idx_employees_position ON employees(position_id);
CREATE INDEX IF NOT EXISTS idx_attendance_employee_date ON attendance(employee_id, work_date);
CREATE INDEX IF NOT EXISTS idx_leave_requests_employee ON leave_requests(employee_id);
CREATE INDEX IF NOT EXISTS idx_payroll_employee_period ON payroll(employee_id, pay_period_start);
CREATE INDEX IF NOT EXISTS idx_payroll_items_payroll ON payroll_items(payroll_id);
CREATE INDEX IF NOT EXISTS idx_work_shifts_employee_date ON work_shifts(employee_id, shift_date);
CREATE INDEX IF NOT EXISTS idx_medical_licenses_employee ON medical_licenses(employee_id);
CREATE INDEX IF NOT EXISTS idx_dept_history_employee ON employee_department_history(employee_id);
CREATE INDEX IF NOT EXISTS idx_pos_history_employee ON employee_position_history(employee_id);
